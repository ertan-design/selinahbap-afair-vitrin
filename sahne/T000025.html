CSS 3B dönme podyumu

Ornek figür panelleri preserve-3d ile bir silindir etrafında döner.

  • CSS 3B dönme (podyum/karusel)
  • Defile podyumu
  • Standart
  • Gece altını

Cihazın hareketi azaltıyor; sahne durağan.

Kaynağı gör
register({
  baslik: 'CSS 3B dönme podyumu',
  aciklama: 'Ornek figür panelleri preserve-3d ile bir silindir etrafında döner.',
  mount: function (host, api) {
    var P = api.palette, N = 8;
    host.style.background = P.bg;
    host.style.overflow = 'hidden';

    var stage = document.createElement('div');
    stage.style.cssText = 'position:absolute;inset:0;display:flex;align-items:center;justify-content:center;perspective:900px;';
    host.appendChild(stage);

    var ring = document.createElement('div');
    ring.style.cssText = 'width:0;height:0;transform-style:preserve-3d;';
    stage.appendChild(ring);

    var caption = document.createElement('div');
    caption.style.cssText = 'position:absolute;left:0;right:0;bottom:12px;text-align:center;font:14px system-ui,sans-serif;color:' + P.fg + ';opacity:.8;';
    caption.textContent = 'Ornek defile: panel sayisi ' + N + ' (ornek veri)';
    host.appendChild(caption);

    var R = 0;

    function build() {
      R = Math.min(api.width, api.height) * 0.3;
      var pw = Math.max(40, 2 * R * Math.sin(Math.PI / N) * 1.08);
      var ph = Math.min(api.height * 0.55, pw * 1.7);
      ring.innerHTML = '';
      for (var i = 0; i < N; i++) {
        var p = document.createElement('div');
        p.style.cssText = 'position:absolute;width:' + pw + 'px;height:' + ph + 'px;margin-left:' + (-pw / 2) + 'px;margin-top:' + (-ph / 2) + 'px;' +
          'transform:rotateY(' + (i * 360 / N) + 'deg) translateZ(' + R + 'px);' +
          'background:linear-gradient(160deg,' + P.a + ',' + P.b + ' 55%,' + P.c + ');' +
          'border:1px solid ' + P.fg + '33;border-radius:6px;display:flex;align-items:center;justify-content:center;' +
          'color:' + P.bg + ';font:600 13px system-ui,sans-serif;backface-visibility:visible;';
        p.textContent = 'Örnek ' + (i + 1);
        ring.appendChild(p);
      }
    }

    build();
    api.onResize(build);

    var stop = api.raf(function () {
      var deg = api.reducedMotion ? 0 : (api.time() / 60) % 360;
      ring.style.transform = 'rotateX(-8deg) rotateY(' + deg + 'deg)';
    });

    return function () {
      stop();
    };
  }
});
