Voronoi hücreleri

Hareketli tohum noktaları; her piksel en yakın tohumun rengini alır. Defile podyumu bağlamı için soyut figür (örnek).

  • Voronoi hücreleri
  • Defile podyumu
  • Standart
  • Gece altını

Cihazın hareketi azaltıyor; sahne durağan.

Kaynağı gör
register({
  baslik: 'Voronoi hücreleri',
  aciklama: 'Hareketli tohum noktaları; her piksel en yakın tohumun rengini alır. Defile podyumu bağlamı için soyut figür (örnek).',
  mount: function (host, api) {
    var c = document.createElement('canvas');
    c.style.cssText = 'display:block;width:100%;height:100%';
    host.appendChild(c);
    var x = c.getContext('2d'), P = api.palette;
    var cols = [P.a, P.b, P.c, P.fg];
    var seed = 7;
    function rnd() {
      seed = (seed * 16807) % 2147483647;
      return (seed - 1) / 2147483646;
    }
    var N = 14, S = [], i;
    for (i = 0; i < N; i++) {
      S.push({
        x: rnd(),
        y: rnd(),
        vx: (rnd() - 0.5) * 0.004,
        vy: (rnd() - 0.5) * 0.004,
        k: i % cols.length
      });
    }
    var W = 2, H = 2, step = 6;
    function fit() {
      c.width = Math.max(2, api.width * api.dpr | 0);
      c.height = Math.max(2, api.height * api.dpr | 0);
      W = c.width;
      H = c.height;
    }
    fit();
    api.onResize(fit);

    function mover() {
      for (var j = 0; j < N; j++) {
        var s = S[j];
        s.x += s.vx;
        s.y += s.vy;
        if (s.x < 0 || s.x > 1) { s.vx = -s.vx; s.x = Math.min(1, Math.max(0, s.x)); }
        if (s.y < 0 || s.y > 1) { s.vy = -s.vy; s.y = Math.min(1, Math.max(0, s.y)); }
      }
    }

    function draw() {
      x.fillStyle = P.bg;
      x.fillRect(0, 0, W, H);
      for (var py = 0; py < H; py += step) {
        for (var px = 0; px < W; px += step) {
          var best = 1e18, bk = 0, j, dx, dy, d;
          for (j = 0; j < N; j++) {
            dx = px - S[j].x * W;
            dy = py - S[j].y * H;
            d = dx * dx + dy * dy;
            if (d < best) { best = d; bk = S[j].k; }
          }
          x.fillStyle = cols[bk];
          x.fillRect(px, py, step, step);
        }
      }
      for (i = 0; i < N; i++) {
        x.fillStyle = P.bg;
        x.beginPath();
        x.arc(S[i].x * W, S[i].y * H, 3 * api.dpr, 0, Math.PI * 2);
        x.fill();
      }
    }

    if (api.reducedMotion) {
      draw();
      return function () {};
    }
    var dur = api.raf(function () {
      mover();
      draw();
    });
    return function () { dur(); };
  }
});
