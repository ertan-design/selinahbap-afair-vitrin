View Transitions: tasarimci duvari

Kart duzeni degisirken document.startViewTransition ile yumusak gecis; destek yoksa dogrudan degisir. [css-view-transitions]

  • View Transitions API
  • Tasarımcı duvarı
  • Standart
  • Şehir neonu

Cihazın hareketi azaltıyor; sahne durağan.

Kaynağı gör
register({
  baslik: 'View Transitions: tasarimci duvari',
  aciklama: 'Kart duzeni degisirken document.startViewTransition ile yumusak gecis; destek yoksa dogrudan degisir. [css-view-transitions]',
  mount: function (host, api) {
    var P = api.palette, doc = host.ownerDocument;
    var root = doc.createElement('div');
    root.style.cssText = 'width:100%;height:100%;box-sizing:border-box;padding:16px;background:' + P.bg + ';color:' + P.fg + ';font:600 14px system-ui,sans-serif;display:flex;flex-direction:column;gap:12px';
    host.appendChild(root);

    var bar = doc.createElement('div');
    bar.style.cssText = 'display:flex;gap:10px;align-items:center;flex-wrap:wrap';
    var btn = doc.createElement('button');
    btn.textContent = 'Duzeni degistir';
    btn.style.cssText = 'background:transparent;color:' + P.a + ';border:1px solid ' + P.a + ';border-radius:8px;padding:6px 12px;cursor:pointer;font:inherit';
    var etiket = doc.createElement('span');
    etiket.style.opacity = '0.8';
    bar.appendChild(btn);
    bar.appendChild(etiket);
    root.appendChild(bar);

    var izgara = doc.createElement('div');
    izgara.style.cssText = 'display:grid;gap:10px;flex:1;min-height:0';
    root.appendChild(izgara);

    var duzen = 0;
    var i, kart;
    for (i = 0; i < 6; i++) {
      kart = doc.createElement('div');
      kart.textContent = 'Ornek tasarimci ' + (i + 1);
      kart.style.cssText = 'border-radius:12px;padding:12px;display:flex;align-items:flex-end;background:linear-gradient(135deg,' + (i % 3 === 0 ? P.a : i % 3 === 1 ? P.b : P.c) + ',' + P.bg + ');color:' + P.fg;
      kart.style.viewTransitionName = 'kart' + i;
      izgara.appendChild(kart);
    }

    function yerlestir() {
      izgara.style.gridTemplateColumns = duzen ? 'repeat(2,1fr)' : 'repeat(3,1fr)';
      etiket.textContent = (duzen ? '2 sutun' : '3 sutun') + ' - ornek veri';
    }

    function uygula() {
      duzen = 1 - duzen;
      if (doc.startViewTransition) {
        doc.startViewTransition(function () { yerlestir(); });
      } else {
        yerlestir();
      }
    }

    api.on(btn, 'click', uygula);
    yerlestir();

    return function () { root.remove(); };
  }
});
